RNF0: Region info datagram

This datagram contains information about one region.

Occurrences

At the end of each region.

Source

  • School detection module.

Contents

struct Rnf0Datagram {
   DatagramHeader header;   // RNF0
   int accepted;            // 1 if this region is accepted, otherwise 0
   int channel;             // The channel this region was detected on
   float threshold;         // The threshold [dB] used for region detection
   BoundingBox boundingBox;
   Values values;
   Statistics statistics;   // Unused since KORONA version 2.15.0
   int borderIdCount;       // Number of border ids
   int borderIds[];         // Ids of Region border datagram for this region
   Histogram histogram;     // Unused since KORONA version 2.15.0
   int perimeterPointCount; // Always 0 since KORONA version 2.15.0
   PerimeterPoint[] perimeterPoints;
   int maskIntervalCount;   // Always 0 prior to KORONA version 2.15.0
   MaskInterval[] maskIntervals;
};

struct BoundingBox {
   float x;      // Ping number relative to beginning of the file
   float y;      // Upper depth [m]
   float width;  // Number of pings
   float height; // Height [m]
};

struct Values {
   float area;            // [m²]
   float sampleCount;     //
   float sumLogSv;        // [dB]
   float sumSv;           // [4π 1852² m²/m³]
   float logMeanSv;       // [dB]
   double sa;             // [m²/nmi²]
   float perimeter;       // [m]
   float length;          // [m]
   float maxHeight;       // [m]
};

struct Statistics {      // Unused since KORONA version 2.15.0
   double meanInside;    // Mean LogSv for samples inside the region close to the border
   double stdDevInside;  // Standard deviation for samples inside the region close to the border
   double meanOutside;   // Mean LogSv for samples outside the region close to the border
   double stdDevOutside; // Standard deviation for samples outside the region close to the border
   double overlap;       // Overlap [%] between the Gauss curves of the two distributions
   double crossing;      // Cross point (between the mean values) of the two distributions
};

struct Histogram {   // Unused since KORONA version 2.15.0
   float startValue; // Start value [dB] of the first histogram bin
   float step;       // Bin size [dB]
   int maxIndex;     // Index of the last bin
   int[] counts;     // Number of samples in each bin
};

struct PerimeterPoint {
   long ntTime;
   float depth;
};

struct MaskInterval {
   long ntTime;
   float minDepth;
   float maxDepth;
};
