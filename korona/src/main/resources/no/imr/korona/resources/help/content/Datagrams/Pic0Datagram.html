This datagram contains all plankton categories referred to later by Plankton inversion data datagrams.

struct Pic0Datagram {
   DatagramHeader header;  // PIC0
   int planktonCategoryCount;
   PlanktonCategory[] planktonCategories;
};

struct PlanktonCategory {
   string name;   // Full name of this plankton category
   string legend; // Short name displayed in e.g. the colorbar
   byte number;
   byte red;
   byte green;
   byte blue;
};
