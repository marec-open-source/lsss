PID0: Plankton inversion data datagram

This datagram contains plankton inversion results with length distribution of the plankton categories. The plankton categories are defined in the Plankton inversion configuration datagram.

Occurrences

Once per ping.

Source

  • Plankton inversion module.

Contents

struct Pid0Datagram {
   DatagramHeader header; // PID0
   int count;             // The number of samples in the original RAW datagram this plankton sample is valid for
   float sampleDistance;  // The distance between the samples [m]
   float firstDepth;      // The depth of the first sample [m]
   int numberOfSamples;   // The number of stored planktonSamples objects
   PlanktonSample[] planktonSamples;
};

struct PlanktonSample {
   int startSampleNumber;     // The first sample number in the original RAW datagram this plankton sample is valid for
   int numberOfPlanktonDatas; // The number of plankton datas for this sample
   PlanktonData[] planktonDatas;
};

struct PlanktonData {
   byte planktonNumber; // Refers to the plankton category number in the Plankton inversion configuration datagram
   int numberOfBinsInLengthDistribution;
   LengthDistribution lengthDistribution; // Only present if numberOfBinsInLengthDistribution > 0
   float fraction; // The fraction of the sA this data unit represents
   float residual; // The computed residual from the inversion algorithm
};

struct LengthDistribution {
   float[] binDividers; // numberOfBinsInLengthDistribution + 1 of these.
   float[] abundances;  // numberOfBinsInLengthDistribution of these.
};
